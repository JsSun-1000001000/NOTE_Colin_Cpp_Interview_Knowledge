要求这样的：输入一个字符串，或者数组，这个你选，咋方便咋来，选的字符串，要求转成链表，然后反转，输出。

```cpp
#include <iostream>
#include <string>

using namespace std;

//链表节点
struct Listnode{
	char val;
	Listnode* next;
	Listnode(char x):val(0),next(nullptr){}
}

class Solution{
public:
	Listnode* stringTolist(const string &s){
		//字符串转换成链表
		if(s.empty()){
			return nullptr;
		}
		Listnode* head = new Listnode(s[0]);
		Listnode* current = head;
		
		for(int i = 1; i < s.length(); i++){
			current->next = new Listnode(s[i]);
			current = current->next;
		}
		return head;
	}
	
	//反转
	Listnode* reverseList(Listnode* head){
		Listnode* pre = nullptr;
		Listnode* current = head;
		Listnode* tmp;
		
		while(current){
			tmp = current->next;
			current->next = pre;
			pre = current;
			cur = tmp;
		}
		return pre;
	}
	
	//打印
	void print(Listnode* head){
		Listnode* current = head;
		while(current){
			cout << current->val;
			if(current->next != nullptr){
				cout << "->";
			}
			current = current->next;
		}
		cout << endl;
	}
}
```
